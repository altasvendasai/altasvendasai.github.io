-- Respostas dos blends ficam no MESMO registro do diagnóstico da cliente
alter table public.acelera_diagnostico add column if not exists blends jsonb;
alter table public.acelera_diagnostico add column if not exists blends_em timestamptz;

-- Abre o link: devolve o nome e o que ela já respondeu (pra continuar de onde parou)
create or replace function public.acelera_blends_abrir(p_token text)
returns jsonb language sql security definer set search_path = public as $$
  select jsonb_build_object(
    'nome', coalesce(previas->>'nome', respostas->>'nome'),
    'respostas', coalesce(blends, '{}'::jsonb),
    'enviado', blends_em is not null)
  from public.acelera_diagnostico where token = p_token;
$$;

-- Salva a cada pergunta; p_final = true quando ela aperta Enviar (depois disso trava)
create or replace function public.acelera_blends_salvar(p_token text, p_respostas jsonb, p_final boolean)
returns boolean language plpgsql security definer set search_path = public as $$
begin
  if length(p_respostas::text) > 20000 then return false; end if;
  update public.acelera_diagnostico
     set blends = p_respostas, blends_em = case when p_final then now() else null end
   where token = p_token and blends_em is null;
  return found;
end $$;

grant execute on function public.acelera_blends_abrir(text) to anon, authenticated;
grant execute on function public.acelera_blends_salvar(text, jsonb, boolean) to anon, authenticated;
